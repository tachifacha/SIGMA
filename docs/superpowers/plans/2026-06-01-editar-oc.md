```php
<?php
session_start();
require "../../../config/conexion.php";

$pdo = db::connect();
$errores = [];
$flash_errors = $_SESSION["flash_errors"] ?? [];
unset($_SESSION["flash_errors"]);
$flash_success = $_SESSION["flash_success"] ?? [];
unset($_SESSION["flash_success"]);
$oc = null;
$detalle = [];
$proveedores = [];

// Cargar proveedores
$proveedores = $pdo
    ->query("SELECT id_proveedor, razon_social FROM proveedores")
    ->fetchAll(PDO::FETCH_ASSOC);

// GET: cargar OC existente
if ($_SERVER["REQUEST_METHOD"] == "GET") {
    if (!isset($_GET["id"])) {
        header("Location: consultarOc.php");
        exit();
    }
    $id = $_GET["id"];

    $stmt = $pdo->prepare("SELECT * FROM ordenes_compra WHERE id_oc = ?");
    $stmt->execute([$id]);
    $oc = $stmt->fetch(PDO::FETCH_ASSOC);

    if (!$oc) {
        $_SESSION["flash_errors"] = ["Orden de compra no encontrada"];
        header("Location: consultarOc.php");
        exit();
    }

    // Cargar detalle con nombres de materiales
    $stmtDet = $pdo->prepare(
        "SELECT od.id_material, od.cantidad, m.nombre
         FROM oc_detalle od
         JOIN materiales m ON od.id_material = m.id_material
         WHERE od.id_oc = ?"
    );
    $stmtDet->execute([$id]);
    $detalle = $stmtDet->fetchAll(PDO::FETCH_ASSOC);
}
?>
```

## Task 2: Crear editarOc.php — Lógica POST (UPDATE + DELETE + INSERT)

**Files:**
- Modify: `src/modulos/compras_provision/ordenes_compra/editarOc.php`

- [ ] **Step 1: Agregar la lógica POST al archivo**

Insertar el bloque POST **antes** del `?>` que cierra el PHP inicial (después del bloque GET del Task 1):

```php
// POST: actualizar OC
if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $id_oc = $_POST["id_oc"];
    $id_proveedor = trim($_POST["id_proveedor"]);
    $fecha_emision = trim($_POST["fecha_emision"]);
    $total_orden = floatval($_POST["total_orden"] ?? 0);
    $materiales_data = json_decode($_POST["materiales_json"] ?? "[]", true);

    if (empty($id_proveedor) || empty($fecha_emision)) {
        $errores[] = "El proveedor y la fecha son obligatorios.";
    }

    if (empty($materiales_data)) {
        $errores[] = "Debe agregar al menos un material.";
    }

    if (empty($errores)) {
        try {
            $pdo->beginTransaction();

            // 1. Actualizar cabecera
            $stmt = $pdo->prepare(
                "UPDATE ordenes_compra SET id_proveedor=?, fecha_emision=?, total=? WHERE id_oc=?"
            );
            $stmt->execute([$id_proveedor, $fecha_emision, $total_orden, $id_oc]);

            // 2. Eliminar detalle existente
            $stmtDel = $pdo->prepare("DELETE FROM oc_detalle WHERE id_oc=?");
            $stmtDel->execute([$id_oc]);

            // 3. Insertar nuevo detalle
            $stmtDet = $pdo->prepare(
                "INSERT INTO oc_detalle (id_oc, id_material, cantidad) VALUES (?, ?, ?)"
            );
            foreach ($materiales_data as $item) {
                $stmtDet->execute([
                    $id_oc,
                    $item["id_material"],
                    $item["cantidad"],
                ]);
            }

            $pdo->commit();
            $_SESSION["flash_success"] = ["Orden de compra actualizada correctamente"];
            header("Location: consultarOc.php");
            exit();
        } catch (PDOException $e) {
            $pdo->rollBack();
            $errores[] = "Error al actualizar orden de compra: " . $e->getMessage();
        }
    }

    if (!empty($errores)) {
        $_SESSION["flash_errors"] = $errores;
        header("Location: editarOc.php?id=" . $id_oc);
        exit();
    }
}
```
